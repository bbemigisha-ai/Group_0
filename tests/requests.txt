##POST request
curl -X POST http://localhost:8080/transactions \
  -u Guest:getout \
  -H "Content-Type: application/json" \
  -d '{
    "categoryID": "PAYMENT",
    "recipientID": 2,
    "senderID": 1,
    "txDate": "27 September 2026",
    "txTime": "14:30:00",
    "updatedBalance": 15000,
    "txAmount": 2000,
    "txFee": 0,
    "currency": "RWF",
    "status": "SUCCESS"
  }'

## PUT request
curl -X PUT http://localhost:8080/transactions/1 \
  -u Guest:getout \
  -H "Content-Type: application/json" \
  -d '{
    "updatedBalance": 18500,
    "txFee": 50,
    "status": "SUCCESS"
  }'
